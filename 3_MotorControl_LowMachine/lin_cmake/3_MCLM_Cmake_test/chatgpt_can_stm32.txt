        }

        case 'F':

            break;

        case 'R':
            msg.type = CMD_REVERSE;
            break;

        case 'X':
            msg.type = CMD_STOP;
            break;

        default:
            // 未知命令 → 保持 CMD_NONE
            break;
    }

    return msg;

    int16_t value;         // 参数（如速度值），对无参数命令无效
} CommandMsg_t;
